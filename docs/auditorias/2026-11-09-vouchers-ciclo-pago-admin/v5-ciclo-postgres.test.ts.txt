async function preparar(t: import("node:test").TestContext): Promise<BaseEfimera | null> {
  if (lista) return lista;

  if (!base) return null;
  apuntarLaAppA(base);

  delete env.ARCA_MODO; // stub: nada sale a la red
  const duenio = await prismaComoDuenio(base);
  const sql = readFileSync(path.join(process.cwd(), "prisma/pending-gate2/CH_voucher_enum.sql"), "utf8");

const hoyDelNegocio = async () => (await import("@/lib/datetime")).todayInBusinessTz();
const clave = (p: string) => `${p}-nf-${String(n++).padStart(4, "0")}-1111-2222-3333`;

/** Un crédito de MP en el extracto (pipeline de mercadopago-auto.ts), «auto» = listo para facturar solo. */
async function movimientoMp(base: BaseEfimera, tenantId: string, monto: number, fecha: string, origen: "mercadopago" | "banco" = "mercadopago") {
  const d = await prismaComoDuenio(base);
  const imp = await d.importacionBancaria.create({
    data: { tenantId, nombreArchivo: `nf-${n++}`, origen, archivo: Buffer.from(""), mapeoJson: {} },
  });
  return (
    await d.movimientoImportado.create({
      data: {
        tenantId,
        importacionId: imp.id,
        hash: `nf:${n++}`,
        fecha: fecha.replace(/-/g, ""),
        monto,
        descripcion: "Transferencia de Adm. La Alameda",
        clasificacion: "venta",
        estadoPropuesta: "auto",
      },
    })
  ).id;
}

async function emitir(tenantId: string) {
  const { emitirPropuestas } = await import("@/lib/bancos-glue");
  return emitirPropuestas(tenantId, "auto");
}

const L = () => import("@/lib/vouchers/liquidacion-repo");

/** Arma el pipeline real del aviso de MP con un clasificador base FACTURABLE y facturación de mentira. */
async function pipelineMp(tenantId: string) {
  const { clasificadorSinPlataDeVouchers, ReconciliacionMovimientosMP } = await import("@/lib/mercadopago-auto");
  const { facturarPagoSiCorresponde } = await import("@/plugins/mercadopago");
  const rec = new ReconciliacionMovimientosMP(tenantId);
  const facturadas: string[] = [];
  const deps = {
    tenantId,
    client: {} as never,
    clasificador: clasificadorSinPlataDeVouchers({ clasificar: async () => ({ clasificacion: "FACTURABLE" as const, motivo: "venta" }) } as never),
    reconciliacion: rec,
    facturar: async (pago: { id: string }) => {
      facturadas.push(pago.id);
      return null as unknown as string; // null = transitorio, no persiste: sólo contamos que llegó a facturar
    },
  };
  const procesar = async (pago: unknown) => {
    rec.recordarPago(pago as never);
    const resumen = { leidos: 0, facturados: 0, noFacturables: 0, aRevisar: 0, rechazados: 0, saltados: 0, errores: 0 };
    await facturarPagoSiCorresponde(pago as never, deps as never, resumen as never);
    return resumen;
  };
  return { procesar, facturadas };
}
const atras = (dia: string, n: number) => { const x = new Date(dia + "T12:00:00Z"); x.setUTCDate(x.getUTCDate() - n); return x.toISOString().slice(0, 10); };


const pagoMp = (id: string, monto: number, fecha: string) => ({
  id, estado: "approved", monto, externalReference: "", operacion: "transferencia",
  fechaAcreditacion: fecha.replace(/-/g, ""), descripcion: "Transferencia recibida", contraparteNombre: "Adm. La Alameda",
});

test("V5-1 · aviso de MP de la transferencia ANTES de registrar el pago (orden natural): ¿se factura?", async (t) => {
  const base = await preparar(t);
  if (!base) return;
  const A = base.a.id;
  const hoy = await hoyDelNegocio();
  const { procesar, facturadas } = await pipelineMp(A);
  await procesar(pagoMp("v5p1", 9191.91, hoy));
  const { registrarPagoACuenta, confirmarAcreditacionMp } = await L();
  const p = await registrarPagoACuenta(A, { clave: clave("v51"), monto: 9191.91, fecha: hoy, medio: "TRANSFERENCIA", actor: "user:test" });
  const c = await confirmarAcreditacionMp(A, { pagoId: p.pagoId, actor: "user:test" });
  console.log("# V5-1 facturadas=", facturadas, "confirmación via", c.via);
  assert.equal(facturadas.length, 0, "la transferencia de la administración llegó a facturarse (pago registrado después del aviso)");
});

test("V5-3 · el mismo crédito en dos filas (aviso + extracto CSV de MP): enganchado uno, ¿se factura el gemelo?", async (t) => {
  const base = await preparar(t);
  if (!base) return;
  const A = base.a.id;
  const d = await prismaComoDuenio(base);
  const hoy = await hoyDelNegocio();
  const { registrarPagoACuenta, confirmarAcreditacionMp } = await L();
  const p = await registrarPagoACuenta(A, { clave: clave("v53"), monto: 9393.93, fecha: atras(hoy, 1), medio: "TRANSFERENCIA", actor: "user:test" });
  const w = await movimientoMp(base, A, 9393.93, hoy);
  const c = await confirmarAcreditacionMp(A, { pagoId: p.pagoId, movimientoId: w, actor: "user:test" });
  assert.equal(c.movimientoId, w);
  const gemelo = await movimientoMp(base, A, 9393.93, hoy); // el extracto CSV de la misma cuenta, importado después
  await emitir(A);
  const m = await d.movimientoImportado.findFirstOrThrow({ where: { id: gemelo } });
  console.log("# V5-3 gemelo:", m.estadoPropuesta, "facturado=", m.invoiceId != null);
  assert.equal(m.invoiceId, null, "el gemelo del crédito enganchado se facturó");
});

test("V5-4 · re-proceso del mapeo de la importación de MP después del enganche (borra y recrea filas): ¿se factura?", async (t) => {
  const base = await preparar(t);
  if (!base) return;
  const A = base.a.id;
  const d = await prismaComoDuenio(base);
  const hoy = await hoyDelNegocio();
  const { registrarPagoACuenta, confirmarAcreditacionMp } = await L();
  const p = await registrarPagoACuenta(A, { clave: clave("v54"), monto: 9494.94, fecha: atras(hoy, 1), medio: "TRANSFERENCIA", actor: "user:test" });
  const w = await movimientoMp(base, A, 9494.94, hoy);
  await confirmarAcreditacionMp(A, { pagoId: p.pagoId, movimientoId: w, actor: "user:test" });
  // confirmarMapeoAction → procesarYPersistir (bancos-actions.ts:232): deleteMany de la importación y createMany de nuevo.
  const viejo = await d.movimientoImportado.findFirstOrThrow({ where: { id: w } });
  await d.movimientoImportado.deleteMany({ where: { id: w } });
  const nuevo = await d.movimientoImportado.create({
    data: { tenantId: A, importacionId: viejo.importacionId, hash: viejo.hash, fecha: viejo.fecha, monto: viejo.monto, descripcion: viejo.descripcion, clasificacion: "venta", estadoPropuesta: "auto" },
  });
  await emitir(A);
  const m = await d.movimientoImportado.findFirstOrThrow({ where: { id: nuevo.id } });
  console.log("# V5-4 recreado:", m.estadoPropuesta, "facturado=", m.invoiceId != null);
  assert.equal(m.invoiceId, null, "tras re-procesar, el crédito ya acreditado se facturó");
});

test("V5-8 · enganchado y después «marcar no facturable» pisa la marca; ¿otro pago igual lo engancha o se factura?", async (t) => {
  const base = await preparar(t);
  if (!base) return;
  const A = base.a.id;
  const d = await prismaComoDuenio(base);
  const hoy = await hoyDelNegocio();
  const { registrarPagoACuenta, confirmarAcreditacionMp } = await L();
  const p1 = await registrarPagoACuenta(A, { clave: clave("v58a"), monto: 9898.98, fecha: atras(hoy, 1), medio: "TRANSFERENCIA", actor: "user:test" });
  const x = await movimientoMp(base, A, 9898.98, hoy);
  await confirmarAcreditacionMp(A, { pagoId: p1.pagoId, movimientoId: x, actor: "user:test" });
  await d.movimientoImportado.update({ where: { id: x }, data: { estadoPropuesta: "no_facturable", motivoRevision: "Marcado como no facturable por el usuario." } });

  console.log("# V5-8 p2 via", c2.via, c2.movimientoId === x ? "ENGANCHÓ EL MISMO" : "");
  assert.notEqual(c2.movimientoId, x, "un movimiento quedó enganchado a dos pagos");
